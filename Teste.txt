  function openTreatmentModal(treatment) {
    const info = treatmentInfo[treatment];
    if (!info) return;
    
    document.getElementById('modalTitle').textContent = info.title;
    document.getElementById('modalBody').innerHTML = info.content;
    document.getElementById('treatmentModal').classList.add('active');
    document.body.style.overflow = 'hidden';
  }

  function closeModal() {
    document.getElementById('treatmentModal').classList.remove('active');
    document.body.style.overflow = '';
  }

  // Close modal when clicking outside
  document.getElementById('treatmentModal').addEventListener('click', function(e) {
    if (e.target === this) {
      closeModal();
    }
  });

  // Close modal with ESC key
  document.addEventListener('keydown', function(e) {
    if (e.key === 'Escape') {
      closeModal();
    }
  });
